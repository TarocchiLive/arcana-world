cmake_minimum_required(VERSION 3.21)
project(arcana_overlay_qt LANGUAGES CXX)

if(APPLE)
    # QCocoaScreen 从 Qt 6.11 起提供稳定的原生显示器映射。
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # 每窗口 layer-shell 集成需要 Qt 6.5。
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets)
else()
    find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets)
endif()
add_library(awqt SHARED overlay.cpp bridge.h platform.h)
target_compile_features(awqt PRIVATE cxx_std_17)
target_compile_definitions(awqt PRIVATE AWQT_BUILD)
target_link_libraries(awqt PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
set_target_properties(awqt PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(awqt PRIVATE platform_mac.mm)
    target_link_libraries(awqt PRIVATE "-framework Cocoa" "-framework CoreGraphics")
    set_target_properties(awqt PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH YES OBJCXX_STANDARD 17)
elseif(WIN32)
    target_sources(awqt PRIVATE platform_windows.cpp)
    set_target_properties(awqt PROPERTIES PREFIX "" IMPORT_PREFIX "lib")
    target_link_libraries(awqt PRIVATE user32)
else()
    find_package(LayerShellQt 6.3 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
    target_link_libraries(awqt PRIVATE LayerShellQt::Interface PkgConfig::WAYLAND_CLIENT)
    target_sources(awqt PRIVATE platform_linux.cpp)
    set_target_properties(awqt PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS awqt LIBRARY DESTINATION . RUNTIME DESTINATION . ARCHIVE DESTINATION .)
